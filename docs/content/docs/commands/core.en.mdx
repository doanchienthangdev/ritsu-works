### `/core diff [doc]`

```bash
node -e "
const fs = require('fs');
const { parse } = require('./scripts/core/lib/frontmatter.cjs');
const { execSync } = require('child_process');
const docArg = process.argv[1];
const slugs = docArg ? [docArg.replace(/\.md$/,'').replace(/^00-core\//,'')] : fs.readdirSync('./00-core').filter(f => f.endsWith('.md') && f !== 'README.md').map(f => f.replace(/\.md$/, ''));
slugs.forEach(slug => {
  const fp = '00-core/' + slug + '.md';
  if (!fs.existsSync(fp)) return;
  const c = fs.readFileSync(fp, 'utf-8');
  const { frontmatter } = parse(c);
  const reviewed = frontmatter.last_reviewed;
  if (!reviewed) { console.log(slug + ': no last_reviewed timestamp — treating as new'); return; }
  const reviewedStr = reviewed instanceof Date ? reviewed.toISOString().slice(0,10) : String(reviewed);
  try {
    const since = execSync('git log -1 --before=\"' + reviewedStr + 'T23:59:59\" --format=%H -- ' + fp, { encoding:'utf-8' }).trim();
    if (!since) { console.log(slug + ': no git commit before ' + reviewedStr); return; }
    console.log('--- ' + slug + ' diff since ' + reviewedStr + ' (' + since.slice(0,7) + ') ---');
    const diff = execSync('git diff ' + since + '..HEAD -- ' + fp, { encoding:'utf-8' });
    console.log(diff || '(no diff)');
  } catch (e) { console.log(slug + ': git error ' + e.message); }
});
" -- <doc>
```

```bash
node -e "
const fs = require('fs');
const path = require('path');
const { serialize } = require('./scripts/core/lib/frontmatter.cjs');
const args = {};
process.argv.slice(1).forEach(a => {
  const m = a.match(/^--([a-z]+)=(.+)$/);
  if (m) args[m[1]] = m[2];
  else if (!args._) args._ = a;
});
const slug = args._;
if (!slug) { console.error('Usage: /core scaffold <doc-slug> --layer=<layer> [--status=stub|v0.1-draft]'); process.exit(1); }
const layer = args.layer;
if (!layer) { console.error('--layer required (identity|strategy|brand-design|operating|stage|meta)'); process.exit(1); }
const status = args.status || 'stub';
const filepath = './00-core/' + slug + '.md';
if (fs.existsSync(filepath)) { console.error(slug + '.md exists. Use /core fill or git rm first.'); process.exit(1); }
const fm = {
  title: slug.replace(/-/g,' ').replace(/\\b\\w/g, c=>c.toUpperCase()),
  type: 'core-doc', slug, layer, status,
  owner: 'founder',
  last_reviewed: new Date().toISOString().slice(0,10),
  review_cadence: 'on-trigger',
  cited_by: [],
  auto_load: false,
};
if (status === 'stub') {
  fm.entry_condition = 'TODO: define entry condition';
  fm.triggered_by = 'TODO: define trigger';
  fm.why_deferred = 'TODO: 1-paragraph why deferred';
} else if (status === 'v0.1-draft') {
  fm.revisit_at = '30-paying';
  fm.revisit_trigger = 'TODO: define revisit trigger';
  fm.revisit_owner = 'founder';
}
const body = '\\n# ' + fm.title + '\\n\\n# Stub — see frontmatter for entry condition\\n';
fs.writeFileSync(filepath, serialize(fm, body), 'utf-8');
```
